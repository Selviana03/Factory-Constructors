ConnectionFactory (Database Pooling) --- class Connection { final String id; Connection(this.id); void execute(String query) => print("Koneksi $id mengeksekusi: $query"); } class ConnectionFactory { // Pooling: menyimpan daftar koneksi yang tersedia static final List _pool = [Connection("DB-1"), Connection("DB-2")]; static int _index = 0; // Mendapatkan koneksi berikutnya dari pool static Connection getConnection() { final conn = _pool[_index]; _index = (_index + 1) % _pool.length; // Mengulang index jika sudah habis return conn; } } // --- 2. NotificationFactory (Subclass selection) --- abstract class Notification { void send(String message); } class EmailNotification implements Notification { @override void send(String msg) => print("Mengirim via Email: $msg"); } class PushNotification implements Notification { @override void send(String msg) => print("Mengirim via Push: $msg"); } class NotificationFactory { // Factory method memilih subclass berdasarkan platform static Notification create(String platform) { if (platform == 'email') return EmailNotification(); if (platform == 'push') return PushNotification(); throw Exception("Platform tidak dikenal"); } } // --- 3. ShapeFactory (Cache untuk objek yang sama) --- class Shape { final String type; Shape(this.type); } class ShapeFactory { static final Map _cache = {}; // Jika sudah ada di cache, gunakan yang ada. Jika belum, buat baru. static Shape getShape(String type) { return _cache.putIfAbsent(type, () => Shape(type)); } } void main() { print("--- Test Latihan 1: Connection Pooling ---"); ConnectionFactory.getConnection().execute("SELECT * FROM users"); ConnectionFactory.getConnection().execute("SELECT * FROM products"); print("\n--- Test Latihan 2: Notification Factory ---"); NotificationFactory.create('email').send('Halo dari Email!'); print("\n--- Test Latihan 3: Shape Factory Cache ---"); var s1 = ShapeFactory.getShape('lingkaran'); var s2 = ShapeFactory.getShape('lingkaran'); print("Apakah objek s1 dan s2 sama? ${identical(s1, s2)}"); // Hasilnya true }
